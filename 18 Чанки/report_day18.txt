3. КОД split_with_overlap()
def split_with_overlap(
    text: str,
    chunk_size: int,
    overlap: int,
    minimum_length: int = 30,
) -> list[str]:
    """Разбиение с перекрытием, обрезкой краёв и отбрасыванием коротких чанков."""
    if chunk_size <= 0:
        raise ValueError("chunk_size должен быть больше 0")
    if overlap < 0:
        raise ValueError("overlap не может быть отрицательным")
    if overlap >= chunk_size:
        raise ValueError("overlap должен быть меньше chunk_size")

    chunks = []
    step = chunk_size - overlap
    for start in range(0, len(text), step):
        chunk = text[start:start + chunk_size].strip()
        if len(chunk) >= minimum_length:
            chunks.append(chunk)
    return chunks

Как работает:
- проверяются три недопустимых значения параметров;
- step = chunk_size - overlap, поэтому следующий чанк начинается раньше
  конца предыдущего;
- срез text[start:start + chunk_size] берёт не больше chunk_size символов;
- strip() убирает пробелы и переводы строк по краям;
- чанк короче minimum_length не добавляется;
- range() идёт от 0 до len(text) с шагом step.

10. ИТОГОВАЯ ПРОГРАММА final_day18.py
Что делает:
1) находит все TXT-файлы в normalized/;
2) читает каждый документ в UTF-8;
3) делит текст на чанки 300/50;
4) проверяет корректность параметров внутри split_with_overlap();
5) удаляет пробелы по краям через strip();
6) не сохраняет пустые и слишком короткие чанки;
7) хранит source, chunk_id и length;
8) создаёт папку chunks/ и сохраняет каждый чанк в TXT;
9) создаёт chunks_summary.csv;
10) выводит число документов, общее число чанков и среднюю длину;
11) проверяет, что нет чанков длиннее 300 и короче 30 символов.

Полный код:

import csv
import os
from pathlib import Path

from chunking import split_with_overlap

os.chdir(os.path.dirname(os.path.abspath(__file__)))

SOURCE_DIR = Path("normalized")
CHUNKS_DIR = Path("chunks")
CSV_FILE = Path("chunks_summary.csv")
CHUNK_SIZE = 300
OVERLAP = 50
MINIMUM_LENGTH = 30

CHUNKS_DIR.mkdir(exist_ok=True)
for old_file in CHUNKS_DIR.glob("*.txt"):
    old_file.unlink()

files = sorted(SOURCE_DIR.glob("*.txt"))
if not files:
    print("В папке normalized нет TXT-файлов.")
    raise SystemExit(1)

all_chunks = []

for file_path in files:
    text = file_path.read_text(encoding="utf-8")
    chunks = split_with_overlap(
        text,
        chunk_size=CHUNK_SIZE,
        overlap=OVERLAP,
        minimum_length=MINIMUM_LENGTH,
    )

    for chunk_id, chunk in enumerate(chunks):
        data = {
            "source": file_path.name,
            "chunk_id": chunk_id,
            "length": len(chunk),
            "text": chunk,
        }
        all_chunks.append(data)

        filename = f"{file_path.stem}_chunk_{chunk_id}.txt"
        (CHUNKS_DIR / filename).write_text(chunk, encoding="utf-8")

with open(CSV_FILE, "w", encoding="utf-8", newline="") as report:
    writer = csv.writer(report)
    writer.writerow(["source", "chunk_id", "length"])
    for chunk in all_chunks:
        writer.writerow([chunk["source"], chunk["chunk_id"], chunk["length"]])

total_length = sum(chunk["length"] for chunk in all_chunks)
average = total_length / len(all_chunks) if all_chunks else 0

print(f"Документов: {len(files)}")
print(f"Всего чанков: {len(all_chunks)}")
print(f"Средняя длина чанка: {round(average, 1)}")

empty_chunks = [chunk for chunk in all_chunks if chunk["text"] == ""]
too_long = [chunk for chunk in all_chunks if chunk["length"] > CHUNK_SIZE]
too_short = [chunk for chunk in all_chunks if chunk["length"] < MINIMUM_LENGTH]

print(f"Пустых чанков: {len(empty_chunks)}")
print(f"Чанков длиннее {CHUNK_SIZE}: {len(too_long)}")
print(f"Чанков короче {MINIMUM_LENGTH}: {len(too_short)}")
print(f"Метаданные и тексты сохранены в {CHUNKS_DIR.name}/ и {CSV_FILE.name}")

15. КОНТРОЛЬНЫЕ ВОПРОСЫ
1. Чанк — небольшой фрагмент исходного документа, который получается при
   разбиении длинного текста.
2. Документы делят на чанки, чтобы искать небольшой подходящий фрагмент,
   передавать модели только релевантную часть, уменьшать лишний контекст
   и точно указывать источник найденной информации.
3. chunk_size — максимальный размер одного фрагмента; сегодня он считается
   в символах.
4. Слишком маленький чанк может потерять смысл: «Python —», «язык»,
   «программирования» по отдельности ничего не объясняют.
5. Слишком большой чанк снова содержит много лишней информации и хуже
   помещается в контекст модели.
6. Overlap — перекрытие соседних чанков, когда конец предыдущего фрагмента
   повторяется в начале следующего.
7. Перекрытие уменьшает риск потерять важную мысль на границе чанков
   и помогает поиску найти информацию на стыке.
8. step = chunk_size - overlap. Например, 300 - 50 = 250.
9. Если overlap >= chunk_size, шаг становится нулевым или отрицательным,
   и цикл не может нормально двигаться по тексту.
10. Если step равен 0, range(0, len(text), 0) выбрасывает ValueError:
    range() arg 3 must not be zero.
11. Фрагмент строки получают срезом: text[start:start + chunk_size].
12. range(0, len(text), step) задаёт индексы начала чанков: от 0 до конца
    текста с шагом step.
13. strip() удаляет пробельные символы в начале и в конце чанка, чтобы
    в файлы не попадали лишние пробелы и переводы строк.
14. Пустые чанки не несут информации, занимают место в CSV и могут
    испортить поиск и подсчёт средней длины.
15. minimum_length отбрасывает слишком короткие «хвосты» последнего чанка,
    в которых нет законченной мысли.
16. source — имя исходного документа, из которого взят чанк.
17. chunk_id — номер чанка внутри документа.
18. enumerate() одновременно даёт номер и сам элемент, поэтому не нужно
    вручную увеличивать счётчик.
19. Длина чанка нужна для контроля качества, статистики и сравнения
    наборов параметров.
20. Path.stem возвращает имя файла без расширения: Path("python.txt").stem
    → "python".
21. Визуальная проверка показывает, что текст читается, слова не разрезаны
    катастрофически, перекрытие работает и важные строки не потеряны.
